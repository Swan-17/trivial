        <button id="reveal-btn" class="btn hidden" onclick="revealInitials()">Reveal Initials</button>

    <script>
        let celebrity = {};
        let currentQuestionIndex = 0;
        let score = 0;
        let answeredCorrectly = false;

        async function fetchCelebrity(category) {
            // Simulate an API call to fetch a celebrity and questions based on category
            // In a real-world scenario, replace this with a real API call
            const response = await fetch(`https://api.example.com/celebrity?category=${category}`);
            const data = await response.json();
            return data;
        }

        async function startQuiz(category) {
            document.getElementById('category-select').classList.add('hidden');
            celebrity = await fetchCelebrity(category);
            currentQuestionIndex = 0;
            score = 0;
            document.getElementById("score").textContent = score;
            loadQuestion();
        }

        function loadQuestion() {
            // Reset state for each question
            answeredCorrectly = false;
            document.getElementById("answer").value = '';
            document.getElementById("submit-btn").disabled = false;

            if (currentQuestionIndex < celebrity.questions.length) {
                document.getElementById("question").textContent = celebrity.questions[currentQuestionIndex].question;
                document.getElementById("skip-btn").classList.remove("hidden");
                document.getElementById("reveal-btn").classList.add("hidden");
            }
        }

        function checkAnswer() {
            const userAnswer = document.getElementById("answer").value.trim().toLowerCase();
            const correctAnswer = celebrity.questions[currentQuestionIndex].answer.toLowerCase();

            if (userAnswer === correctAnswer) {
                score += (5 - currentQuestionIndex) * 10; // Points based on question number
                document.getElementById("score").textContent = score;
                answeredCorrectly = true;
                alert('Correct!');
            } else {
                alert('Incorrect! Try the next question.');
            }
            currentQuestionIndex++;
            if (currentQuestionIndex >= celebrity.questions.length) {
                document.getElementById("reveal-btn").classList.remove("hidden");
                document.getElementById("skip-btn").classList.add("hidden");
            }
            loadQuestion();
        }

        function skipQuestion() {
            if (currentQuestionIndex < celebrity.questions.length) {
                currentQuestionIndex++;
                loadQuestion();
            }
        }

        function revealInitials() {
            alert("The celebrity's initials are: " + celebrity.initials);
        }
    </script>
