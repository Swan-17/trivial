    <script>
        // ⚠️ HARDCODED API KEY (Replace or keep short-term as discussed)
        const GEMINI_API_KEY = AQ.Ab8RN6KnQJrm9a5foD_wJDHBcJNMDKmuuHkujeXuLRzbSve3Rw; 
        
        let appState = {
            topic: "",
            wikiText: "",
            exampleQuestions: [],
            fullQuiz: [],
            currentIndex: 0,
            score: 0,
            selectedDifficulty: ""
        };

        function switchView(viewName) {
            ['viewTopic', 'viewLoading', 'viewPreview', 'viewQuiz', 'viewResults'].forEach(v => {
                document.getElementById(v).classList.add('hidden');
            });
            document.getElementById(viewName).classList.remove('hidden');
        }

        function updateStepIndicator(text) {
            document.getElementById('stepIndicator').innerText = text;
        }

        // --- STEP 1: Wikipedia Scraper Engine ---
        async function fetchWikipediaData(topic) {
            try {
                // 1. Search for page title
                const searchUrl = `https://en.wikipedia.org/w/api.php?action=query&origin=*&format=json&list=search&srsearch=${encodeURIComponent(topic)}&srlimit=3`;
                const searchRes = await fetch(searchUrl);
                const searchData = await searchRes.json();
                
                if (!searchData.query.search.length) throw new Error("Topic not found on Wikipedia.");
                
                const mainTitle = searchData.query.search[0].title;
                const secondTitle = searchData.query.search[1] ? searchData.query.search[1].title : null;

                // 2. Fetch main page content and subtopic references
                const contentUrl = `https://en.wikipedia.org/w/api.php?action=query&origin=*&format=json&prop=extracts|links&exintro=true&explaintext=true&titles=${encodeURIComponent(mainTitle)}${secondTitle ? '|' + encodeURIComponent(secondTitle) : ''}`;
                const contentRes = await fetch(contentUrl);
                const contentData = await contentRes.json();
                
                const pages = contentData.query.pages;
                let combinedText = "";
                let linksList = [];

                for (let pageId in pages) {
                    if (pages[pageId].extract) {
                        combinedText += `--- Topic: ${pages[pageId].title} ---\n` + pages[pageId].extract + "\n\n";
                    }
                    if (pages[pageId].links) {
                        // grab up to 15 sub-links for extra context flavor
                        linksList.push(...pages[pageId].links.slice(0, 15).map(l => l.title));
                    }
                }

                return {
                    combinedText,
                    subtopics: [...new Set(linksList)].slice(0, 20).join(", ")
                };
            } catch (err) {
                console.error(err);
                throw err;
            }
        }

        // --- GEMINI API HELPER ---
        async function callGemini(promptText) {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${GEMINI_API_KEY}`;
            const response = await fetch(url, {
                method: "POST",
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: promptText }] }],
                    generationConfig: { responseMimeType: "application/json" }
                })
            });
            const data = await response.json();
            if (data.error) throw new Error(data.error.message);
            return JSON.parse(data.candidates[0].content.parts[0].text);
        }

        // --- HANDLER: Fetch Preview & 5 Questions ---
        async function handleFetchPreview() {
            const topic = document.getElementById('topicInput').value.trim();
            if (!topic) return alert("Please enter a valid topic.");

            appState.topic = topic;
            switchView('viewLoading');
            updateStepIndicator('Scraping Wiki...');
            document.getElementById('loadingText').innerText = `Crawling Wikipedia for "${topic}" and subtopics...`;

            try {
                const wikiData = await fetchWikipediaData(topic);
                appState.wikiText = wikiData.combinedText;

                document.getElementById('loadingText').innerText = `Analyzing content & generating 5 preview questions via AI...`;

                const prompt = `Based on the following Wikipedia data for "${topic}" (and subtopics: ${wikiData.subtopics}), generate a JSON array of exactly 5 multiple choice questions. 
                Progress them in difficulty: Question 1 must be very easy (Level 1), Question 2 easy (Level 2), Question 3 moderate (Level 3), Question 4 hard (Level 4), and Question 5 expert/niche (Level 5).
                Return strict JSON format matching this schema:
                [
                  {
                    "level": 1,
                    "question": "...",
                    "options": ["A", "B", "C", "D"],
                    "correct": 0
                  }
                ]
                Wiki Data: ${appState.wikiText.substring(0, 8000)}`;

                appState.exampleQuestions = await callGemini(prompt);

                // Populate UI
                document.getElementById('wikiSummaryContainer').innerText = appState.wikiText.substring(0, 500) + "...";
                
                let listHtml = "";
                appState.exampleQuestions.forEach(q => {
                    listHtml += `
                        <div class="bg-slate-800 border border-slate-700 p-3 rounded-xl text-xs space-y-1">
                            <span class="text-indigo-400 font-bold">Level ${q.level}:</span>
                            <p class="font-medium text-slate-200">${q.question}</p>
                        </div>
                    `;
                });
                document.getElementById('exampleQuestionsList').innerHTML = listHtml;

                switchView('viewPreview');
                updateStepIndicator('Step 2: Choose Difficulty');
            } catch (err) {
                alert("Error: " + err.message);
                switchView('viewTopic');
            }
        }

        // --- HANDLER: Start Full 20-Question Quiz ---
        async function handleStartFullQuiz() {
            appState.selectedDifficulty = document.getElementById('difficultySelect').value;
            switchView('viewLoading');
            document.getElementById('loadingText').innerText = `Generating full 20-question quiz at tier: ${appState.selectedDifficulty}...`;

            try {
                const prompt = `Based on the Wikipedia context provided, generate a JSON array of exactly 20 multiple choice questions suited for difficulty: "${appState.selectedDifficulty}".
                Return strict JSON format matching this schema:
                [
                  {
                    "question": "...",
                    "options": ["A", "B", "C", "D"],
                    "correct": 0
                  }
                ]
                Wiki Context: ${appState.wikiText.substring(0, 10000)}`;

                appState.fullQuiz = await callGemini(prompt);
                appState.currentIndex = 0;
                appState.score = 0;

                switchView('viewQuiz');
                updateStepIndicator('Step 3: Quiz Active');
                renderQuizQuestion();
            } catch (err) {
                alert("Error generating full quiz: " + err.message);
                switchView('viewPreview');
            }
        }

        // --- QUIZ GAMEPLAY LOGIC ---
        function renderQuizQuestion() {
            const q = appState.fullQuiz[appState.currentIndex];
            document.getElementById('quizProgressText').innerText = `Question ${appState.currentIndex + 1} of ${appState.fullQuiz.length}`;
            document.getElementById('quizScoreText').innerText = `Score: ${appState.score}`;
            document.getElementById('questionTitle').innerText = q.question;
            document.getElementById('nextQuestionBtn').classList.add('hidden');

            let optionsHtml = "";
            q.options.forEach((opt, idx) => {
                optionsHtml += `
                    <button onclick="handleSelectOption(this, ${idx}, ${q.correct})" 
                        class="option-btn w-full text-left bg-slate-800 hover:bg-slate-700 border border-slate-700 rounded-xl p-4 transition font-medium text-slate-200">
                        ${opt}
                    </button>
                `;
            });
            document.getElementById('optionsContainer').innerHTML = optionsHtml;
        }

        function handleSelectOption(btnElement, selectedIdx, correctIdx) {
            const buttons = document.querySelectorAll('.option-btn');
            buttons.forEach(b => b.disabled = true); // Disable further clicks

            if (selectedIdx === correctIdx) {
                btnElement.classList.remove('bg-slate-800', 'border-slate-700');
                btnElement.classList.add('bg-emerald-900/50', 'border-emerald-500', 'text-emerald-200');
                appState.score++;
            } else {
                btnElement.classList.remove('bg-slate-800', 'border-slate-700');
                btnElement.classList.add('bg-rose-900/50', 'border-rose-500', 'text-rose-200');
                // Highlight correct option
                buttons[correctIdx].classList.remove('bg-slate-800', 'border-slate-700');
                buttons[correctIdx].classList.add('bg-emerald-900/50', 'border-emerald-500', 'text-emerald-200');
            }

            document.getElementById('quizScoreText').innerText = `Score: ${appState.score}`;
            document.getElementById('nextQuestionBtn').classList.remove('hidden');
        }

        function handleNextQuestion() {
            appState.currentIndex++;
            if (appState.currentIndex < appState.fullQuiz.length) {
                renderQuizQuestion();
            } else {
                switchView('viewResults');
                updateStepIndicator('Complete');
                document.getElementById('finalScoreDisplay').innerText = `${appState.score} / ${appState.fullQuiz.length}`;
            }
        }
